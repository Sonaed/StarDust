cmake_minimum_required(VERSION 3.16)
project(StellarDustCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(StellarDustCore STATIC src/stardust_core.cpp)
target_include_directories(StellarDustCore PUBLIC include)
add_executable(stardust-core src/stardust_core_cli.cpp)
target_link_libraries(stardust-core PRIVATE StellarDustCore)
